$$
\begin{aligned}
\operatorname{VIH}_{effective}
  &= \frac{\sum_{i=1}^{n} \mathbf{1}[\operatorname{verified}(i)] \cdot (1 - \operatorname{staleness}(c_i)) \cdot \operatorname{quality}(i)}{\operatorname{planningLatency} + \operatorname{modelLatency} + \operatorname{verificationLatency} + \operatorname{mergeLatency}} \\
  &\leq \frac{n}{\operatorname{planningLatency} + \operatorname{modelLatency} + \operatorname{verificationLatency} + \operatorname{mergeLatency}}
\end{aligned}
$$
